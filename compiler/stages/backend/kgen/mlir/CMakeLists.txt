find_package(LLVM 18.1 CONFIG QUIET PATHS /usr/lib/llvm-18/lib/cmake/llvm)
find_package(MLIR CONFIG QUIET PATHS /usr/lib/llvm-18/lib/cmake/mlir)
if(LLVM_FOUND)
  message(STATUS "A KGEN MLIR backend: LLVM ${LLVM_PACKAGE_VERSION} found (MLIR ${MLIR_FOUND})")
  add_library(a_kgen_mlir STATIC kgen_mlir_codegen.cpp)
  target_include_directories(a_kgen_mlir PRIVATE ${LLVM_INCLUDE_DIRS} ${CMAKE_SOURCE_DIR}/compiler ${CMAKE_SOURCE_DIR}/compiler/stages ${CMAKE_SOURCE_DIR}/runtime)
  target_compile_definitions(a_kgen_mlir PRIVATE A_HAVE_KGEN_LLVM=1 ${LLVM_DEFINITIONS})
  llvm_map_components_to_libnames(A_KGEN_LLVM_LIBS core support target native)
  target_link_libraries(a_kgen_mlir PUBLIC a_type_checker a_diagnostics ${A_KGEN_LLVM_LIBS})
  add_library(a::kgen_mlir ALIAS a_kgen_mlir)
else()
  message(STATUS "A KGEN MLIR backend: LLVM dev packages not found; using fallback")
  add_library(a_kgen_mlir INTERFACE)
  add_library(a::kgen_mlir ALIAS a_kgen_mlir)
endif()
